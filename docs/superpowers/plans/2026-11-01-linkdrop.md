# LinkDrop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build LinkDrop — a watermark-free media downloader ecosystem that functions as a high-speed Telegram Community Bot with native 60s slow-mode countdown and dynamic sponsor ad captions, paired with a modern Next.js 14+ web downloader and admin dashboard deployed to Cloudflare.

**Architecture:** Unified Next.js 14+ App Router project deployed to Cloudflare Edge. Incoming Telegram group messages are processed via `/api/telegram/webhook`, where the multi-engine extractor (Cobalt + TikWM + IG fallbacks) resolves clean CDN media URLs and passes them directly to Telegram's `sendVideo` API alongside active sponsor copy stored in Cloudflare KV. The web frontend provides a sleek, dark obsidian UI using the Manrope font for manual downloads and admin management.

**Tech Stack:** Next.js 14+, TypeScript, Tailwind CSS, Manrope (Google Fonts), Lucide Icons, Telegram Bot API, Cobalt REST API v10, TikWM API, Cloudflare Pages/Workers (`@opennextjs/cloudflare`).

```ts
import { SupportedPlatform } from "../types";

export function detectPlatform(url: string): SupportedPlatform {
  try {
    const parsed = new URL(url);
    const host = parsed.hostname.toLowerCase();

    if (host.includes("tiktok.com")) return "tiktok";
    if (host.includes("instagram.com")) return "instagram";
    if (host.includes("youtube.com") || host.includes("youtu.be")) return "youtube";
    if (host.includes("snapchat.com")) return "snapchat";
    if (host.includes("facebook.com") || host.includes("fb.watch")) return "facebook";
    if (host.includes("twitter.com") || host.includes("x.com")) return "twitter";
    if (host.includes("reddit.com")) return "reddit";

    return "unknown";
  } catch {
    return "unknown";
  }
}

export function extractUrlFromText(text: string): string | null {
  const urlRegex = /(https?:\/\/[^\s]+)/g;
  const matches = text.match(urlRegex);
  return matches && matches.length > 0 ? matches[0] : null;
}
```

```ts
import { ExtractedMedia } from "../types";

export async function extractTikTok(url: string): Promise<ExtractedMedia | null> {
  try {
    const res = await fetch("https://www.tikwm.com/api/", {
      method: "POST",
      headers: {
        "Content-Type": "application/x-www-form-urlencoded; charset=UTF-8",
        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
      },
      body: new URLSearchParams({ url, count: "12", cursor: "0", web: "1", hd: "1" }),
    });

    const data = await res.json();
    if (data.code !== 0 || !data.data) {
      return null;
    }

    const d = data.data;
    const cleanVideoUrl = d.hdplay || d.play;

    return {
      platform: "tiktok",
      originalUrl: url,
      title: d.title || "TikTok Video",
      author: d.author?.unique_id || d.author?.nickname,
      thumbnail: d.cover,
      duration: d.duration,
      downloadUrl: cleanVideoUrl,
      audioUrl: d.music,
      options: [
        {
          quality: "HD (No Watermark)",
          format: "mp4",
          url: cleanVideoUrl,
        },
        ...(d.music
          ? [
              {
                quality: "Audio Original (MP3)",
                format: "mp3" as const,
                url: d.music,
                isAudioOnly: true,
              },
            ]
          : []),
      ],
    };
  } catch (err) {
    console.error("TikWM extraction error:", err);
    return null;
  }
}
```

- [ ] **Step 4: Create `lib/extractor/cobalt.ts` (Universal No-Watermark Engine)**

```ts
import { ExtractedMedia, SupportedPlatform } from "../types";

const PUBLIC_COBALT_INSTANCES = [
  "https://api.cobalt.tools",
  "https://cobalt-api.kwiatekm.tokyo",
  "https://cobalt.api.timelessoses.world",
];

export async function extractCobalt(
  url: string,
  platform: SupportedPlatform
): Promise<ExtractedMedia | null> {
  for (const instance of PUBLIC_COBALT_INSTANCES) {
    try {
      const response = await fetch(`${instance}/`, {
        method: "POST",
        headers: {
          Accept: "application/json",
          "Content-Type": "application/json",
          "User-Agent": "LinkDrop/1.0",
        },
        body: JSON.stringify({
          url,
          videoQuality: "1080",
          filenameStyle: "pretty",
          downloadMode: "auto",
        }),
      });

      if (!response.ok) continue;

      const data = await response.json();
      const streamUrl = data.url;

      if (streamUrl) {
        return {
          platform,
          originalUrl: url,
          title: `${platform.toUpperCase()} Video (Clean)`,
          downloadUrl: streamUrl,
          options: [
            {
              quality: "HD 1080p (No Watermark)",
              format: "mp4",
              url: streamUrl,
            },
          ],
        };
      }
    } catch {
      continue;
    }
  }
  return null;
}
```

```ts
import { ExtractedMedia } from "../types";
import { detectPlatform } from "./platform";
import { extractTikTok } from "./tikwm";
import { extractCobalt } from "./cobalt";

export async function extractMedia(url: string): Promise<ExtractedMedia> {
  const platform = detectPlatform(url);

  if (platform === "tiktok") {
    const tikwmResult = await extractTikTok(url);
    if (tikwmResult) return tikwmResult;
  }

  // Use Cobalt for all supported platforms (or as fallback)
  const cobaltResult = await extractCobalt(url, platform);
  if (cobaltResult) return cobaltResult;

  throw new Error(`Could not extract clean media from ${platform || "this link"}. The link may be private or unsupported.`);
}
```

```ts
import { NextRequest, NextResponse } from "next/server";

export const runtime = "edge";

export async function GET(req: NextRequest) {
  const searchParams = req.nextUrl.searchParams;
  const targetUrl = searchParams.get("url");
  const filename = searchParams.get("filename") || "linkdrop-video.mp4";

  if (!targetUrl) {
    return new NextResponse("Missing url parameter", { status: 400 });
  }

  try {
    const mediaRes = await fetch(targetUrl);
    if (!mediaRes.ok) {
      return new NextResponse("Failed to fetch source media", { status: mediaRes.status });
    }

    const headers = new Headers();
    headers.set("Content-Type", mediaRes.headers.get("Content-Type") || "video/mp4");
    headers.set("Content-Disposition", `attachment; filename="${filename}"`);
    if (mediaRes.headers.get("Content-Length")) {
      headers.set("Content-Length", mediaRes.headers.get("Content-Length")!);
    }

    return new NextResponse(mediaRes.body, { headers });
  } catch (error) {
    return new NextResponse("Stream proxy error", { status: 500 });
  }
}
```

```ts
import { SponsorAdSettings } from "../types";

const TELEGRAM_API_BASE = "https://api.telegram.org";

export class TelegramBotClient {
  private token: string;

  constructor(token?: string) {
    this.token = token || process.env.TELEGRAM_BOT_TOKEN || "";
  }

  async sendVideo(chatId: number | string, videoUrl: string, caption: string, replyToMessageId?: number, sponsorAd?: SponsorAdSettings) {
    const inlineKeyboard = sponsorAd && sponsorAd.enabled ? [
      [
        { text: sponsorAd.buttonText || "🚀 Check Sponsor", url: sponsorAd.clickUrl || "https://t.me" }
      ]
    ] : [];

    const res = await fetch(`${TELEGRAM_API_BASE}/bot${this.token}/sendVideo`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        chat_id: chatId,
        video: videoUrl,
        caption: caption,
        parse_mode: "HTML",
        reply_to_message_id: replyToMessageId,
        reply_markup: inlineKeyboard.length > 0 ? { inline_keyboard: inlineKeyboard } : undefined,
      }),
    });

    return await res.json();
  }

  async sendMessage(chatId: number | string, text: string, replyToMessageId?: number) {
    const res = await fetch(`${TELEGRAM_API_BASE}/bot${this.token}/sendMessage`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        chat_id: chatId,
        text,
        parse_mode: "HTML",
        reply_to_message_id: replyToMessageId,
      }),
    });
    return await res.json();
  }

  async setChatSlowModeDelay(chatId: number | string, seconds: number = 60) {
    const res = await fetch(`${TELEGRAM_API_BASE}/bot${this.token}/setChatSlowModeDelay`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        chat_id: chatId,
        slow_mode_delay: seconds,
      }),
    });
    return await res.json();
  }
}
```

```ts
import { NextResponse } from "next/server";
import { extractUrlFromText } from "@/lib/extractor/platform";
import { extractMedia } from "@/lib/extractor";
import { TelegramBotClient } from "@/lib/telegram/bot";
import { getSponsorSettings, incrementDownloadStats } from "@/lib/storage/kv";

export const runtime = "edge";

export async function POST(req: Request) {
  try {
    const update = await req.json();
    const message = update.message;

    if (!message || !message.text) {
      return NextResponse.json({ ok: true });
    }

    const chatId = message.chat.id;
    const messageId = message.message_id;
    const text = message.text;

    // Check if message contains a link
    const url = extractUrlFromText(text);
    if (!url) {
      if (text === "/start" || text === "/help") {
        const bot = new TelegramBotClient();
        await bot.sendMessage(
          chatId,
          `👋 <b>Welcome to LinkDrop!</b>\n\nSimply paste any TikTok, Instagram, YouTube, or Snapchat link in this chat to download it instantly without watermark.\n\n⏳ <i>Group Slow Mode: 60s per download</i>`,
          messageId
        );
      }
      return NextResponse.json({ ok: true });
    }

    const bot = new TelegramBotClient();
    const adSettings = await getSponsorSettings();

    try {
      const media = await extractMedia(url);
      await incrementDownloadStats(media.platform);

      let caption = `✨ <b>Downloaded via LinkDrop (No Watermark)</b>\n\n`;
      if (media.title) {
        caption += `🎬 <i>${media.title.substring(0, 80)}</i>\n\n`;
      }

      if (adSettings.enabled) {
        caption += `📢 <b>${adSettings.title}</b>\n${adSettings.body}\n<a href="${adSettings.clickUrl}">👉 Click Here</a>\n\n`;
      }

      caption += `⏳ <i>Slow mode active: 60s cooldown</i>`;

      const sent = await bot.sendVideo(chatId, media.downloadUrl, caption, messageId, adSettings);

      // Fallback if video was >50MB and Telegram rejected upload
      if (!sent.ok && sent.description?.includes("too big")) {
        await bot.sendMessage(
          chatId,
          `⚠️ <b>File exceeds Telegram 50MB direct limit.</b>\n\n<a href="${media.downloadUrl}">⬇️ Click here to download directly in HD</a>`,
          messageId
        );
      }
    } catch (err: any) {
      await bot.sendMessage(
        chatId,
        `❌ <b>Could not download:</b> ${err.message || "Failed to extract clean media."}`,
        messageId
      );
    }

    return NextResponse.json({ ok: true });
  } catch (error) {
    return NextResponse.json({ ok: true });
  }
}
```
